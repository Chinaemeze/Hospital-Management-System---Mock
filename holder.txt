import initialState from './initialstate';
import useHospitalDashboard from './CustomHooks/HospitalDashboardHook';
import DispatchContext from './ContextFiles/DispatchContext';

import AllDoctorsContext from './ContextFiles/AllDoctorsContext';

Reducer.jsx:-

function reducer(state, action) {
    switch (action.type) {
        case 'SELECT_PATIENT':
            return {
                ...state,
                selectedPatientId: action.id
            }

        case 'ADD_PATIENT':
            return {
                ...state,
                patients: [...state.patients, action.new_patient_object]
            }

        case 'DELETE_PATIENT':
            return {
                ...state,
                selectedPatientId: null,
                patients: state.patients.filter(pat => pat.id !== action.patient_id)
            }

        case 'CHANGE_STATUS':
            return {
                ...state,
                patients: state.patients.map(pat => {
                    if (pat.id === action.patient_id) {
                        return {
                            ...pat,
                            status: action.new_status
                        }
                    }

                    return pat
                })
            }

        case 'CHANGE_PRIORITY':
            return {
                ...state,
                patients: state.patients.map(pat => {
                    if (pat.id === action.patient_id) {
                        return {
                            ...pat,
                            priority: action.new_priority
                        }
                    }

                    return pat
                })
            }

        case 'ASSIGN_DOCTOR':
            return {
                ...state,
                patients: state.patients.map(pat => {
                    if (pat.id === action.patient_id) {
                        return {
                            ...pat,
                            doctorId: action.new_doctor_id
                        }
                    }

                    return pat
                })
            }

        case 'ADD_NOTE':
            return {
                ...state,
                patients: state.patients.map(pat => {
                    if (pat.id === action.patient_id) {
                        return {
                            ...pat,
                            notes: [...pat.notes, action.new_note_object]
                        }
                    }

                    return pat
                })
            }

        case 'UPDATE_VITALS':
            return {
                ...state,
                patients: state.patients.map(pat => {
                    if (pat.id === action.patient_id) {
                        return {
                            ...pat,
                            vitals: {
                                ...pat.vitals,
                                [action.key_of_concern]: action.new_value
                            }
                        }
                    }

                    return pat
                })
            }

        case 'SET_SEARCH':
            return {
                ...state,
                search: action.value
            }

        case 'SET_STATUS_FILTER':
            return {
                ...state,
                statusFilter: action.value
            }

        case 'SET_PRIORITY_FILTER':
            return {
                ...state,
                priorityFilter: action.value
            }

        case 'SET_DOCTOR_FILTER':
            return {
                ...state,
                doctorFilter: action.value
            }

        default:
            return state
    }
}

export default reducer;

import reducer from "../reducer/ReducerFunction";
import { useMemo, useReducer } from "react";

function useHospitalDashboard(initial_stateful_data) {
    // initialise state with reducer
    const [generalState, dispatch] = useReducer(reducer, initial_stateful_data);



    const selectedPatient = useMemo(() => {
        return generalState.patients.find(pat => {
            return pat.id === generalState.selectedPatientId
        })
    }, [generalState.selectedPatientId, generalState.patients]);

    // FIRST PHASE FILTERING
    const phase1FilteredBySearch = useMemo(() => {

        return generalState.patients.filter(pat => {
            if (!generalState.search) return true
            const doctorOfPatient = generalState.doctors.find(doc => doc.id === pat.doctorId);

            const docHasName = doctorOfPatient.name.toLowerCase().replace(/\s+/g, "").includes(generalState.search.toLowerCase().replace(/\s+/g, ""));

            const docHasSpecialty = doctorOfPatient.specialty.toLowerCase().includes(generalState.search.toLowerCase().replace(/\s+/g, ""));


            const nameMatch = pat.name.toLowerCase().includes(generalState.search.toLowerCase().replace(/\s+/g, ""));

            return nameMatch || docHasName || docHasSpecialty

        })
    }, [generalState.search, generalState.patients]);



    // SECOND PHASE FILTERING
    const phase2FilteredByOptions = useMemo(() => {
        // check to see if all filters === 'all'
        const filtersSetAsAll = (generalState.statusFilter === 'all') && (generalState.priorityFilter === 'all') && (generalState.doctorFilter === 'all');
        

        return phase1FilteredBySearch.filter(pat => {
            if (filtersSetAsAll) return true

            

            return (
                ((generalState.statusFilter === 'all') ? true : (generalState.statusFilter === pat.status))
                &&
                ((generalState.doctorFilter === 'all') ? true : (Number(generalState.doctorFilter) === pat.doctorId))
                &&
                ((generalState.priorityFilter === 'all') ? true : (generalState.priorityFilter === pat.priority))
            )
        })
    }, [generalState.priorityFilter, generalState.statusFilter, generalState.doctorFilter, phase1FilteredBySearch]);

    // derived statistics
    const patientsStatisticsObject = useMemo(() => {
        return generalState.patients.reduce((acc, pat) => {
            if(pat.priority === 'critical') {
                acc[pat.priority] = acc[pat.priority] || 0;
                acc[pat.priority] +=1;
            }

            if(pat.status === 'waiting' || pat.status === 'discharged') {
                acc[pat.status] = acc[pat.status] || 0;
                acc[pat.status] +=1
            }

            acc['totalNumberOfPatients'] = generalState.patients.length

            return acc

        }, {});

    }, [generalState.patients, selectedPatient])

    const allDoctors = useMemo(() => {
        return generalState.doctors
    }, [])

    const allPriorities = useMemo(() => {
        return generalState.patients.reduce((acc, pat) => {
            if(!acc.includes(pat.priority)) {
                acc.push(pat.priority)
            }
            return acc
        }, [])
    }, [])


    //exposed values
    return {
        generalState,
        selectedPatient,
        dispatch,
        phase2FilteredByOptions,
        patientsStatisticsObject,
        allDoctors,
        allPriorities
    }
}

export default useHospitalDashboard;

import { memo, useCallback, useRef } from "react";

const AddNewPatientForm = memo(function ({ all_doctors, dispatch, all_priorities }) {
    const generateThreeDigitId = useCallback(() => {
        return Math.floor(Math.random() * 900) + 100;
    }, []);

    const newPatientObjectRef = useRef({
        status: 'waiting'
    });

    const updateValueInRef = useCallback((target, key_in_ref) => {
        if (key_in_ref === 'vitals') {
            newPatientObjectRef.current[key_in_ref] = newPatientObjectRef.current[key_in_ref] || {};

            if (target.id === 'heartRate') {
                newPatientObjectRef.current[key_in_ref][target.id] = Number(target.value);
            } else {
                newPatientObjectRef.current[key_in_ref][target.id] = target.value
            }

            return
        }

        if (key_in_ref === 'age' || key_in_ref === 'doctorId') {
            newPatientObjectRef.current[key_in_ref] = Number(target.value);
            console.log(newPatientObjectRef.current)
            return
        }

        newPatientObjectRef.current[key_in_ref] = target.value;

        // text log
        console.log(newPatientObjectRef.current)
    }, []);

    const addPatientInGeneralState = useCallback(() => {
        if (Object.keys(newPatientObjectRef.current).length !== 6) return

        newPatientObjectRef.current['notes'] = [];
        newPatientObjectRef.current['id'] = generateThreeDigitId()

        const objectToSend = { ...newPatientObjectRef.current }

        dispatch({
            type: 'ADD_PATIENT',
            new_patient_object: objectToSend
        })

        newPatientObjectRef.current = {
            status: 'waiting'
        }
    }, [])

    const handleKeyDown = useCallback((e) => {
        const allowed = ['1', '2', '3', '4', '5', '6', '7', '8', '9', '0', '/', 'Backspace'];

        if(!allowed.includes(e.key)) {
            e.preventDefault()
        }
        
    }, [])




    return (
        <div>
            <h3>Add New Patient</h3>
            {/* for name */}
            <div>
                <label htmlFor="new-patient-name">Enter Patient's Name</label>
                <input type="text" id="new-patient-name" placeholder="Enter new patient's name" data-key="name" onChange={(e) => {
                    updateValueInRef(e.target, e.target.dataset.key)
                }} />
            </div>
            {/* for age */}
            <div>
                <label htmlFor="new-patient-age">Enter Age</label>
                <input type="number" id="new-patient-age" placeholder="Input patient's age" data-key="age" onChange={(e) => {
                    updateValueInRef(e.target, e.target.dataset.key)
                }} />
            </div>
            {/* for vitals */}
            <div onChange={e => { updateValueInRef(e.target, e.target.dataset.key) }} >
                <label htmlFor="heartRate">Enter HeartRate</label>
                <input type="number" data-key="vitals" placeholder="Input Heart Rate" id="heartRate" />
                <br />
                <label htmlFor="bloodPressure">Enter Blood Pressure</label>
                <input type="text" id="bloodPressure" data-key="vitals" placeholder="Input blood pressure" onKeyDown={e => handleKeyDown(e)} />
            </div>
            {/* for doctor id select  */}
            <div>
                <h4>Assign Doctor</h4>
                <div onChange={e => {
                    updateValueInRef(e.target, e.target.dataset.key)
                }}>
                    {
                        all_doctors.map(doc => {
                            return (
                                <div key={doc.id} >
                                    <label htmlFor={doc.name}>{doc.name}</label>
                                    <input type="radio" name="doctors" id={doc.name} value={doc.id} data-key="doctorId" />
                                </div>
                            )

                        })
                    }
                </div>
            </div>
            {/* For priority */}
            <div>
                <h4>Assign Priority</h4>
                <div onChange={e => {
                    updateValueInRef(e.target, e.target.dataset.key)
                }}>
                    {
                        all_priorities.map((priority, ind) => {
                            return (
                                <div key={ind} >
                                    <label htmlFor={priority}>{priority}</label>
                                    <input type="radio" name="priority" id={priority} value={priority} data-key="priority" />
                                </div>
                            )

                        })
                    }
                </div>
            </div>
            <button onClick={addPatientInGeneralState} >Add Patient</button>

        </div>
    );
})

        });

                    updateSelectedVitals(e.target.value)

import { memo, useContext, useMemo } from "react";
import DispatchContext from "../../../ContextFiles/DispatchContext";
import GeneralStateContext from "../../../ContextFiles/GeneralStateContext";
import AllDoctorsContext from "../../../ContextFiles/AllDoctorsContext";
import NewPrioritySelect from "./ActionPanelComponents/NewPrioritySelect";
import NewStatusSelect from "./ActionPanelComponents/NewStatusSelect";
import NewDoctorSelect from "./ActionPanelComponents/NewDoctorSelect";

const ActionPanel = memo(function ({ id_of_patient }) {
    const dispatch = useContext(DispatchContext);
    const generalState = useContext(GeneralStateContext);
    const allDoctors = useContext(AllDoctorsContext);


    const allOptions = useMemo(() => {
        return generalState.patients.reduce((acc, pat) => {
            acc['statuses'] = acc['statuses'] || [];
            acc['priorities'] = acc['priorities'] || [];

            if (!acc['doctors']) {
                acc['doctors'] = allDoctors;
            }

            if (!acc['statuses'].includes(pat.status)) {
                acc['statuses'].push(pat.status)
            }

            if (!acc['priorities'].includes(pat.priority)) {
                acc['priorities'].push(pat.priority)
            }

            return acc

        }, {})
    }, [])

    return (
        <div>
            <h5>Actions</h5>
            <div>
                <div>
                    <NewPrioritySelect id={id_of_patient} options={allOptions.priorities} dispatch={dispatch} />
                    <NewStatusSelect id={id_of_patient} options={allOptions.statuses} dispatch={dispatch} />
                    <NewDoctorSelect id={id_of_patient} list_of_doctors={allOptions.doctors} dispatch={dispatch} />
                </div>
                <button onClick={() => {
                    dispatch({
                        type: 'DELETE_PATIENT',
                        patient_id: id_of_patient
                    })
                }}>Delete Patient {id_of_patient}</button>
            </div>
        </div>
    )
})

        <div>
            <DoctorFilterSelector doctors_list={doctors} dispatch={dispatch} />
            <PriorityFilterSelector priorities_list={priorities} dispatch={dispatch} />
            <StatusFilterSelector statuses_list={statuses} dispatch={dispatch} />

        </div>
    );
}

export default FilterPanel;

        </div>
    )
})
